function closeMoveInspectionModal() {
  const modal = document.getElementById('moveInspectionModal');
  if (modal) modal.classList.add('hidden');
}

function syncHUD() {
  const h = HEROES[currentHero];
  const isLeader = h.isLeader;
  const remPoints = getPointsRemaining();
  const lvl = isMilestoneMode ? 10 : 2;

  // Header party tabs active state & color contrast
  const btnAbra = document.getElementById('btnPartyAbra');
  const btnPancham = document.getElementById('btnPartyPancham');
  if (currentHero === 'abra') {
    btnAbra.className = "skew-panel px-4 py-1.5 bg-[#f5f6f8] text-[#08101a] border-2 border-cyan-400 font-bold text-sm shadow-[0_0_12px_rgba(0,216,246,0.35)] hover:scale-105 transition-all relative";
    btnPancham.className = "skew-panel px-4 py-1.5 bg-[#142334] text-neutral-300 border-2 border-black font-bold text-sm hover:scale-105 transition-all relative";
  } else {
    btnAbra.className = "skew-panel px-4 py-1.5 bg-[#142334] text-neutral-300 border-2 border-black font-bold text-sm hover:scale-105 transition-all relative";
    btnPancham.className = "skew-panel px-4 py-1.5 bg-[#f5f6f8] text-[#08101a] border-2 border-cyan-400 font-bold text-sm shadow-[0_0_12px_rgba(0,216,246,0.35)] hover:scale-105 transition-all relative";
  }

  // Level Up notification badges in party tabs (Hide when allocated/seen)
  const abraLvBadge = document.getElementById('abraLvUpBadge');
  if (abraLvBadge) {
    abraLvBadge.classList.toggle('hidden', !heroLevelUpNotifs.abra);
  }
  const panchamLvBadge = document.getElementById('panchamLvUpBadge');
  if (panchamLvBadge) {
    panchamLvBadge.classList.toggle('hidden', !heroLevelUpNotifs.pancham);
  }

  // Hero Identification Card
  document.getElementById('heroRoleLabel').textContent = isLeader ? 'PROTAGONIST / LEADER' : 'TACTICAL ALLY (AI)';
  document.getElementById('heroSpeciesSub').textContent = h.species;
  document.getElementById('heroNameHeading').textContent = h.name;
  document.getElementById('heroLevelNumber').textContent = lvl;
  document.getElementById('heroNextExpNumber').textContent = `${calcNextExp(lvl)} EXP`;

  // Nature (D-393)
  document.getElementById('heroNatureName').textContent = h.nature.name.toUpperCase();
  document.getElementById('heroNatureModifier').textContent = `(${h.nature.modText})`;
  document.getElementById('heroNatureTactic').textContent = h.nature.tacticText;

  // Badges per D-413: 1 wide block if mono-element, 2 blocks if dual-element
  const badgesContainer = document.getElementById('heroElementBadges');
  badgesContainer.innerHTML = '';
  if (h.elements.length === 1) {
    const elData = ELEMENTS_12.find(e => e.id === h.elements[0]) || ELEMENTS_12[0];
    badgesContainer.innerHTML = `
      <div onclick="explainElement('${elData.id}')" class="h-8 px-3.5 flex items-center justify-center font-mono font-black text-xs rounded neo-border cursor-pointer hover:scale-105 active:scale-95 transition-all" style="background-color: ${elData.bg}; color: ${elData.fg};" title="Click for ${elData.name} rules">
        ${elData.name
